### 起始代码

```cpp
#include <stdexcept>
#include <string>

class Joint {
public:
    Joint(std::string name, double angle)
        : name_{/* 补全 */}, angle_{/* 补全 */} {
        // 建立不变量
    }

    double angle() const { return angle_; }

private:
    std::string name_;
    double angle_{};
};
```
